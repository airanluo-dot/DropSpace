cmake_minimum_required(VERSION 3.25)
project(dropspace_language_worker LANGUAGES CXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  message(FATAL_ERROR "This worker requires Apple Silicon macOS")
endif()
if(NOT DEFINED FASTTEXT_SOURCE OR NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "Pinned FASTTEXT_SOURCE and LLAMA_SOURCE checkouts are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0")
set(CMAKE_OSX_ARCHITECTURES arm64)
# Avoid upstream -march=native and its executable main in the library target.
set(ENGINE_SOURCES args autotune densematrix dictionary fasttext loss matrix meter model productquantizer quantmatrix utils vector)
foreach(SOURCE IN LISTS ENGINE_SOURCES)
  list(APPEND ENGINE_FILES "${FASTTEXT_SOURCE}/src/${SOURCE}.cc")
endforeach()
add_library(fasttext-native STATIC ${ENGINE_FILES})
target_include_directories(fasttext-native PUBLIC "${FASTTEXT_SOURCE}/src")
add_executable(language-worker main.cpp)
target_include_directories(language-worker PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(language-worker PRIVATE fasttext-native)
set_target_properties(language-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
