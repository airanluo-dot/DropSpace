cmake_minimum_required(VERSION 3.19)
project(dropspace_plain_lyrics LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "LLAMA_SOURCE must point to the verified pinned source tree")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(plain-lyrics-worker main.cpp)
target_compile_features(plain-lyrics-worker PRIVATE cxx_std_17)
target_link_libraries(plain-lyrics-worker PRIVATE llama-common)
if(GGML_VULKAN)
  find_package(Vulkan REQUIRED)
  target_link_libraries(plain-lyrics-worker PRIVATE Vulkan::Vulkan)
  target_compile_definitions(plain-lyrics-worker PRIVATE DROPSPACE_VULKAN=1)
endif()
if(MSVC)
  target_compile_options(plain-lyrics-worker PRIVATE /utf-8)
endif()
set_target_properties(plain-lyrics-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC AND GGML_VULKAN)
  # Delay-load and explicitly preload the driver loader from System32, never the cache/CWD.
  target_link_options(plain-lyrics-worker PRIVATE /DELAYLOAD:vulkan-1.dll)
  target_link_libraries(plain-lyrics-worker PRIVATE delayimp)
endif()
