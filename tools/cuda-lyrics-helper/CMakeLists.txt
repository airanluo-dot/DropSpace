cmake_minimum_required(VERSION 3.24)
project(dropspace_cuda_lyrics LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE OR NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  message(FATAL_ERROR "Verified LLAMA_SOURCE and explicit CUDA architectures are required")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LLAMA_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE engine_commit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LLAMA_SOURCE}" status --porcelain --untracked-files=all
  OUTPUT_VARIABLE engine_changes COMMAND_ERROR_IS_FATAL ANY)
if(NOT engine_commit STREQUAL "7fe450e19305b828c199d602c23a8337aaa1f03b" OR NOT engine_changes STREQUAL "")
  message(FATAL_ERROR "CUDA experiment requires the clean pinned engine")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(CUDAToolkit 12.3 REQUIRED)
if(NOT CUDAToolkit_VERSION VERSION_LESS 13)
  message(FATAL_ERROR "Experiment v1 requires CUDA 12; a new ABI needs its own reviewed component")
endif()
set(GGML_CUDA ON CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_STATIC ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
foreach(backend HIP VULKAN SYCL OPENCL METAL BLAS RPC MUSA WEBGPU OPENVINO HEXAGON ZENDNN VIRTGPU VIRTGPU_BACKEND)
  set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
foreach(feature TESTS EXAMPLES TOOLS SERVER APP UI BORINGSSL LIBRESSL)
  set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature OPENSSL SUBPROCESS LLGUIDANCE)
  set(LLAMA_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
set(worker_source "${CMAKE_CURRENT_SOURCE_DIR}/../plain-lyrics-helper/main.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${worker_source}" "${CMAKE_CURRENT_SOURCE_DIR}/adapt_worker.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/adapt_worker.py"
  --source "${worker_source}" --output "${CMAKE_BINARY_DIR}/cuda-worker.cpp"
  COMMAND_ERROR_IS_FATAL ANY)
add_executable(plain-lyrics-worker-cuda "${CMAKE_BINARY_DIR}/cuda-worker.cpp")
target_compile_features(plain-lyrics-worker-cuda PRIVATE cxx_std_17)
target_include_directories(plain-lyrics-worker-cuda PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../plain-lyrics-helper")
target_link_libraries(plain-lyrics-worker-cuda PRIVATE llama-common CUDA::cudart_static)
if(MSVC)
  target_compile_options(plain-lyrics-worker-cuda PRIVATE /utf-8)
endif()
set_target_properties(plain-lyrics-worker-cuda PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
