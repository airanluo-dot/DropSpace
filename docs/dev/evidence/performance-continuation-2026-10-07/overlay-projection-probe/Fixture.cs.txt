// Event producer fixture, not a text measurement approximation. It emits one
// externally supplied new size; the real Window ApplySnapshot decides when this
// invalidation is already consumed and when a follow-up projection is needed.
class FakeCompact : FakeElement
{
    private readonly MediaRenderQueue _queue = new();
    public double IdealIslandWidth = 280, IdealIslandHeight = 40;
    public event EventHandler? IdealWidthChanged;
    public bool ThrowOnRefresh;
    private double _nextWidth, _nextHeight;
    public int Preparations;
    public void ChangeOnPreparation(double width, double height)
    {
        _nextWidth = width; _nextHeight = height;
        _queue.Request(false);
    }
    public override void RefreshForPresentation()
    {
        _queue.Cancel();
        if (!_queue.BeginImmediateRender()) return;
        Preparations++;
        if (ThrowOnRefresh) throw new InvalidOperationException("fixture event producer");
        IdealIslandWidth = _nextWidth; IdealIslandHeight = _nextHeight;
        IdealWidthChanged?.Invoke(this, EventArgs.Empty);
    }
    public void SetAvailableWidth(double width) { }
}

class DispatcherQueue
{
    public static readonly DispatcherQueue Shared = new();
    private readonly Queue<Action> _pending = new();
    public int Scheduled;
    public static DispatcherQueue GetForCurrentThread() => Shared;
    public bool TryEnqueue(Action callback) { Scheduled++; _pending.Enqueue(callback); return true; }
    public int Pending => _pending.Count;
    public void Reset() { _pending.Clear(); Scheduled = 0; }
    public void Drain() { while (_pending.TryDequeue(out var callback)) callback(); }
}
